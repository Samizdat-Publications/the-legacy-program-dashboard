## Global constraints (every task implicitly includes these)

- **Sources.** No invented sources, URLs, document numbers or quotations, anywhere, including placeholder
  and test data (`CLAUDE.md` section 2). A registry or topic may only carry a URL that already exists in
  `research/`.
- **No em dashes** (U+2014) in anything written: code, comments, UI strings, docs, commits, chat. Use a
  comma, colon, full stop or spaced hyphen. Verbatim quotations of sources keep their own characters.
- **Visual rules:** size never encodes importance; one label size (11px); labels chosen by space, never
  shrunk or truncated; no anonymous marks.
- **Checks:** every new check ships with a poison that damages the REAL page (not a predicate or
  fixture) and must make that check fail. `npm run check:poison` proves it.
- **Presentation never moves evidence.** Tiers are never changed by presentation work; the AARO/DoD cap
  stays.
- **The dataset is untouched.** No authored research value is rewritten to fit the design. New
  structure goes in new files or side-cars. The only edits to authored data are integrity repairs that
  restore something the pipeline dropped, and display-only label normalisation (phase 3).
- **Canonical path:** `C:\Users\stewa\ClaudeProjects\Missing individuals connected to defense contractors\the-program`.
  Never work in the OneDrive copy (dead cloud placeholders).
- **Git:** one branch per phase, `redesign/pN-<slug>`. Merge to master only with the full gate, the app
  checks and every poison green. If the harness refuses a push to master (it counts as a production
  deploy), stop and ask Stewart in one line.
- **Windows:** the interpreter is `python`; `core.autocrlf` is on, so anything that hashes bytes
  normalises CRLF; quote every path. Every command block written for Stewart is PowerShell, starts with
  its own quoted `cd`, and has one command per line.
- **Phones:** every page added or rebuilt from phase 1 on works at 375px wide, with no horizontal page
  scroll.

```ts
/* vocab.ts: reader names for the board's twelve edge types, in reading order.
 *
 * The backbone comes first (what the record says a thing is and did), then the
 * assertion layer (who says so, who denies it), then context. The split is
 * measured, not chosen: notes/redesign-2026-09-24/B-GRAPH-STRUCTURE.md section 4.
 * app/checks/run.mjs restates these names in C22 rather than importing them.
 * Phase 3 moves this table into schema/vocab.json, read by the pipeline too.
 */
export type Layer = 'backbone' | 'assertion' | 'context';
export interface EdgeTypeName { id: string; label: string; blurb: string; layer: Layer; }

export const EDGE_TYPES: EdgeTypeName[] = [
  { id: 'custody', label: 'Custody', blurb: 'who is said to hold material, or to have handed it on', layer: 'backbone' },
  { id: 'funding', label: 'Money', blurb: 'who paid, contracted or was paid', layer: 'backbone' },
  { id: 'programmatic', label: 'Programme links', blurb: 'what ran, produced, succeeded or absorbed what', layer: 'backbone' },
  { id: 'institutional', label: 'Institutional ties', blurb: 'posts, roles, memberships and parent bodies', layer: 'backbone' },
  { id: 'legislative', label: 'Law and oversight', blurb: 'what created, governs, amended or must report to what', layer: 'backbone' },
  { id: 'research', label: 'Research and authorship', blurb: 'who studied, wrote or co-wrote what', layer: 'backbone' },
  { id: 'investigative', label: 'Investigations', blurb: 'who examined, analysed or looked into what', layer: 'assertion' },
  { id: 'witness-chain', label: 'Testimony', blurb: 'who said what about whom, and how it reached the record', layer: 'assertion' },
  { id: 'behavioral', label: 'Conduct', blurb: 'actions and moves, including career moves and reprisal', layer: 'assertion' },
  { id: 'denial', label: 'Denials', blurb: 'dated, attributable statements that something is not so', layer: 'assertion' },
  { id: 'temporal', label: 'Sequence in time', blurb: 'what followed, answered or was caused by what', layer: 'context' },
  { id: 'geographic', label: 'Place', blurb: 'where something was, happened or was based', layer: 'context' },
];

export const LAYER_HEAD: Record<Layer, string> = {
  backbone: 'What the record says it is and did',
  assertion: 'Who says so, and who denies it',
  context: 'Time and place',
};

export const NO_SENTENCE = 'no sentence recorded for this link';

export function edgeTypeName(id: string): EdgeTypeName {
  return EDGE_TYPES.find((t) => t.id === id) || { id, label: id, blurb: '', layer: 'context' };
}
```

```ts
/* connections.ts: a node's sourced connections, grouped by relationship and by
 * counterpart, each link with its own sentence, dates, tier and sources.
 *
 * One row per counterpart per relationship: a neighbour joined by two
 * institutional edges is one row with two sentences, not two rows.
 * Order: relationships in the reading order of vocab.ts; rows by earliest date,
 * then strongest tier, then label.
 */
import { h, mark } from '../dom';
import { clsColour, dateRange, type Connection } from '../data';
import { EDGE_TYPES, LAYER_HEAD, NO_SENTENCE, edgeTypeName, type Layer } from '../vocab';

export interface ConnRow { id: string; label: string; cls: string; links: Connection[]; }
export interface ConnGroup { type: string; rows: ConnRow[]; }

const ORDER = EDGE_TYPES.map((t) => t.id);
const rank = (type: string): number => { const i = ORDER.indexOf(type); return i < 0 ? 99 : i; };
const firstDate = (r: ConnRow): string => r.links.map((l) => l.dateStart || '9999').sort()[0];
const bestTier = (r: ConnRow): number => Math.min(...r.links.map((l) => l.claimTier ?? 3));

export function groupConnections(conns: Connection[]): ConnGroup[] {
  const byType = new Map<string, Map<string, ConnRow>>();
  for (const c of conns) {
    let g = byType.get(c.type);
    if (!g) { g = new Map(); byType.set(c.type, g); }
    let r = g.get(c.id);
    if (!r) { r = { id: c.id, label: c.label, cls: c.cls, links: [] }; g.set(c.id, r); }
    r.links.push(c);
  }
  return [...byType.keys()]
    .sort((a, b) => rank(a) - rank(b) || a.localeCompare(b))
    .map((type) => ({
      type,
      rows: [...byType.get(type)!.values()].sort((a, b) =>
        firstDate(a).localeCompare(firstDate(b)) || bestTier(a) - bestTier(b) || a.label.localeCompare(b.label)),
    }));
}

function rowEl(r: ConnRow, type: string): HTMLElement {
  const row = h('div', { class: 'conn' + (type === 'denial' ? ' denial' : ''), 'data-id': r.id });
  row.append(h('a', { class: 'conn-head', href: '#/n/' + r.id }, mark(clsColour(r.cls)), h('span', { class: 'row-label' }, r.label)));
  for (const l of r.links) {
    row.append(h('div', { class: 'conn-line' },
      h('span', { class: 'conn-text' }, l.edgeLabel || l.note || NO_SENTENCE),
      h('span', { class: 'cmeta' }, dateRange(l.dateStart, l.dateEnd) + ' · T' + (l.claimTier ?? '?')),
    ));
    const src = l.sources || [];
    if (src.length) {
      const d = h('details', { class: 'conn-src' }, h('summary', null, src.length + ' source' + (src.length === 1 ? '' : 's')));
      for (const s of src) {
        const text = (s.name || s.url || '') + (s.date ? ' (' + s.date + ')' : '');
        d.append(s.url ? h('a', { href: s.url, target: '_blank', rel: 'noreferrer noopener' }, text) : h('span', null, text));
      }
      row.append(d);
    }
  }
  return row;
}

export function renderConnGroups(groups: ConnGroup[]): HTMLElement {
  const wrap = h('div', { class: 'conn-groups' });
  let layer: Layer | null = null;
  for (const g of groups) {
    const t = edgeTypeName(g.type);
    if (t.layer !== layer) {
      layer = t.layer;
      wrap.append(h('div', { class: 'conn-layer eyebrow dim' }, LAYER_HEAD[layer]));
    }
    const box = h('div', { class: 'conn-group', 'data-type': g.type, 'data-layer': t.layer },
      h('div', { class: 'conn-ghead' },
        h('span', { class: 'eyebrow' + (g.type === 'denial' ? ' red' : '') }, t.label), ' ',
        h('span', { class: 'dim' }, '· ' + g.rows.length),
```

```python
"""lint_no_em_dash.py: fail if U+2014 appears in the app's or the Worker's source.

Stewart's standing rule (global CLAUDE.md): no em dashes anywhere. Data is out of
scope here: verbatim quotations keep their characters, and display labels are
normalised in the pipeline (phase 3). app/src/analyst/tools.ts is exempt until
phase 12, because C19 parses its output and a change is a Worker redeploy.

  python tools/lint_no_em_dash.py              exit 1 on any hit
  python tools/lint_no_em_dash.py --self-test  shows the lint rejecting a planted dash
"""
import os, sys, tempfile

ROOT = os.path.abspath(os.path.join(os.path.dirname(os.path.abspath(__file__)), ".."))
DIRS = ["app/src", "app/checks", "worker/src"]
EXEMPT = {os.path.normpath("app/src/analyst/tools.ts")}
EXTS = (".ts", ".mjs", ".js", ".css", ".html")
DASH = "\u2014"


def hits(root: str) -> list[str]:
    out = []
    for d in DIRS:
        base = os.path.join(root, d)
        for dp, _, fns in os.walk(base):
            for fn in fns:
                if not fn.endswith(EXTS):
                    continue
                p = os.path.join(dp, fn)
                rel = os.path.normpath(os.path.relpath(p, root))
                if rel in EXEMPT:
                    continue
                with open(p, encoding="utf-8") as f:
                    for i, line in enumerate(f, 1):
                        if DASH in line:
                            out.append(f"{rel}:{i}")
    return out


def self_test() -> int:
    with tempfile.TemporaryDirectory() as t:
        os.makedirs(os.path.join(t, "app", "src"))
        with open(os.path.join(t, "app", "src", "x.ts"), "w", encoding="utf-8") as f:
            f.write("const a = 'one " + DASH + " two';\n")
        got = hits(t)
    ok = got == [os.path.normpath("app/src/x.ts") + ":1"]
    print(("ok    " if ok else "FAIL  ") + "the lint rejects a planted em dash :: " + repr(got))
    return 0 if ok else 1


if __name__ == "__main__":
    if "--self-test" in sys.argv:
        sys.exit(self_test())
    h = hits(ROOT)
    for x in h:
        print("em dash: " + x)
    print(("FAIL  " if h else "ok    ") + f"no em dash in {', '.join(DIRS)} :: {len(h)} line(s)")
    sys.exit(1 if h else 0)
```
